# Chargement de healthcare dans mongodb avec arrêt de mongodb
docker compose up --build --abort-on-container-exit

# Chargement de healthcare dans mongodb sans arrêt de mongodb
docker compose up -d --build

# Démarrage de mongodb seul
docker compose up -d --build mongodb

# Lancer le shell mongodb
docker exec -it mongodb mongosh -u root -p "$env:MONGO_INITDB_ROOT_PASSWORD" --authenticationDatabase admin

# Exemple d'utilisation de mongosh (javascript)
use healthcare
db.patients.countDocuments()
db.patients.getIndexes()
db.patients.findOne()

Conteneurs en cours:	docker ps

Stack Compose:			docker compose ps
